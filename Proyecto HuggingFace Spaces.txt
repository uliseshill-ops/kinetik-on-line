Para no alterar nada de la lógica en main.py, dejamos la misma firma de función generar_storytelling_mexicano. Usaremos el endpoint compatible con OpenAI o la API gratuita de Groq (requiere tu variable de entorno o API key gratuita):

import os
import requests

# Puedes obtener una API Key gratuita en https://console.groq.com/
GROQ_API_KEY = os.getenv("GROQ_API_KEY", "tu_api_key_aqui")

def generar_storytelling_mexicano(nombre_producto, descripcion_corta, estado_origen):
    prompt = f"""
    Eres un experto en marketing internacional y cultura mexicana. 
    Producto: {nombre_producto}
    Origen: {estado_origen}
    Descripción: {descripcion_corta}

    Genera dos historias cautivadoras para el producto.
    
    ESTRUCTURA REQUERIDA:
    [ESPAÑOL]
    Crea una descripción emocional y profesional de 2 párrafos que resalte 
    la calidad de lo 'Hecho en México' y la tradición de {estado_origen}.
    
    [ENGLISH]
    (Translate the same story into professional, high-end marketing English)
    
    REGLAS:
    - No uses más de 4 párrafos en total.
    - Mantén el tono artesanal y premium.
    """
    
    # Endpoint público y ultra-rápido de Groq para Llama 3
    url = "https://api.groq.com/openai/v1/chat/completions"
    headers = {
        "Authorization": f"Bearer {GROQ_API_KEY}",
        "Content-Type": "application/json"
    }
    payload = {
        "model": "llama-3.3-70b-versatile",
        "messages": [
            {"role": "user", "content": prompt}
        ],
        "temperature": 0.7
    }
    
    try:
        response = requests.post(url, json=payload, headers=headers, timeout=10)
        if response.status_code == 200:
            return response.json()['choices'][0]['message']['content']
        else:
            return f"Hecho en México: {nombre_producto} de alta calidad. (IA offline temporalmente)."
    except Exception as e:
        return f"Prenda artesanal mexicana premium. ({nombre_producto})"
